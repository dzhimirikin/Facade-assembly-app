/* ===================================================== LOAD FILTERED DATA ===================================================== */ const filteredData = JSON.parse(sessionStorage.getItem("filteredRows") || "[]"); /* ===================================================== ELEMENTS ===================================================== */ const container = document.getElementById("detailContainer"); /* ===================================================== RENDER PDF ===================================================== */ function renderPdf(mode = "facade") { if (!filteredData.length) { container.innerHTML = `

No filtered data

`; return; } container.innerHTML = `
`; // Group data by facade or element let groupedData; if (mode === "facade") { groupedData = filteredData.reduce((acc, item) => { if (!acc[item.facadeId]) acc[item.facadeId] = []; acc[item.facadeId].push(item); return acc; }, {}); } else { groupedData = filteredData.reduce((acc, item) => { const key = `${item.facadeId}.${item.subElementId}`; if (!acc[key]) acc[key] = []; acc[key].push(item); return acc; }, {}); } for (const [groupKey, rows] of Object.entries(groupedData)) { const printSheet = document.createElement("div"); printSheet.className = "print-sheet"; // Header printSheet.innerHTML = `

ASSEMBLY REPORT

${mode === "facade" ? `

${groupKey}

` : `

${rows[0].subElementId} фасада ${rows[0].facadeId}

`}`; rows.forEach(item => { // Format timestamp let tsStr = ""; try { if (item.timestamp?.seconds) { tsStr = new Date(item.timestamp.seconds * 1000).toLocaleString(); } else if (item.timestamp) { tsStr = new Date(item.timestamp).toLocaleString(); } } catch { tsStr = item.timestamp || ""; } const photos = (item.photos || []).filter(p => p); // Operation table const tableHtml = `
Операция${item.stage || ""}
Исполнитель${item.employee || ""}
Дата${tsStr}
Примечание${item.note || ""}
`; // Photo section const photosHtml = photos.length ? `
${photos.map(photo => `
${typeof photo === "string" ? `
${photo}
` : `` }
`).join("")}
` : ""; printSheet.innerHTML += tableHtml + photosHtml + `
`; }); container.appendChild(printSheet); } } /* ===================================================== INIT ===================================================== */ window.addEventListener("DOMContentLoaded", () => { // Add mode selection buttons const toolbar = document.createElement("div"); toolbar.className = "print-toolbar"; toolbar.innerHTML = ` `; container.parentNode.insertBefore(toolbar, container); document.getElementById("facadeBtn").addEventListener("click", () => renderPdf("facade")); document.getElementById("elementBtn").addEventListener("click", () => renderPdf("element")); });
